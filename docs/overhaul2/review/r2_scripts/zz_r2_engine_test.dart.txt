const _config = '{"initialStats": {"charm": 10}, "actions": []}';
const _chars = '''[
  {"id": "a", "name": "가람", "gender": "f", "role": "senior"},
  {"id": "b", "name": "나래", "gender": "m", "role": "senior"}
]''';
const _endings =
    '[{"id": "solo", "name": "솔로", "tier": "solo", "default": true}]';

Map<String, dynamic> _ev(
  String id, {
  String layer = 'daily',
  int? day,
  Map<String, dynamic>? trigger,
  List<Map<String, dynamic>>? lines,
  List<List<Map<String, dynamic>>>? variants,
  List<Map<String, dynamic>>? choices,
  String? cliffhanger,
  String? clock,
  bool? once,
}) => {
  'id': id,
  'layer': layer,
  'day': ?day,
  'trigger': ?trigger,
  'cliffhanger': ?cliffhanger,
  'clock': ?clock,
  'once': ?once,
  'lines':
      lines ??
      [
        {'who': 'them', 'text': '안녕'},
      ],
  if (variants != null)
    'variants': [
      for (final v in variants) {'lines': v},
    ],
  'choices':
      choices ??
      [
        {'text': '하나'},
        {'text': '둘'},
      ],
};

StoryBundle _bundle(
  List<Map<String, dynamic>> events, {
  String? starts,
  bool? deep,
}) => StoryBundle.fromJsonStrings(
  config: _config,
  characters: _chars,
  events: [jsonEncode(events)],
  endings: _endings,
  starts: starts,
  deep: deep,
);

Future<GameController> _ctl(StoryBundle b, {Map<String, Object>? prefs}) async {
  SharedPreferences.setMockInitialValues(prefs ?? {});
  final c = GameController(bundle: b, save: SaveService());
  await c.init();
  await c.markIntroSeen();
  return c;
}

Future<void> _startDay(GameController c) async {
  c.beginMorning();
  c.state!.hearts = c.config.maxHearts;
  await c.startDay(const DayAction(id: 'x', name: 'x', desc: ''));
}

String _rd(String f) => File('assets/story/$f').readAsStringSync();

  group('chain cliffhanger', () {
    Future<String?> run(StoryBundle b, int Function(GameController) pick, {bool Function(GameController)? undoAt}) async {
      final c = await _ctl(b);
      await c.newGame(seed: 5);
      await _startDay(c);
      final order = <String>[];
      while (c.phase == Phase.event) {
        order.add(c.current!.id);
        c.choose(pick(c));
        if (undoAt != null && undoAt(c)) {
          c.undoChoice();
          c.choose(1);
        }
        c.continueAfterChoice();
      }
      print('order=$order cliff=${c.cliffhanger}');
      return c.cliffhanger;
    }

    test('multi-hop main -> daily -> daily(cliff) beats main', () async {
      final b = _bundle([
        _ev('m1', layer: 'main', day: 1, cliffhanger: 'M', choices: [
          {'text': 'go', 'next': 'h1'},
          {'text': 'no'},
        ]),
        _ev('h1', trigger: {
          'day': [0, 0],
        }, choices: [
          {'text': 'go', 'next': 'h2'},
          {'text': 'no'},
        ]),
        _ev('h2', cliffhanger: 'H2', trigger: {
          'day': [0, 0],
        }),
        _ev('o', cliffhanger: 'O', trigger: {
          'day': [1, 1],
        }),
      ]);
      expect(await run(b, (c) => 0), 'H2');
    });

    test('failNext chain from main', () async {
      final b = _bundle([
        _ev('m1', layer: 'main', day: 1, cliffhanger: 'M', choices: [
          {'text': 'go', 'chance': 0, 'failNext': 'f1', 'next': 'h1'},
          {'text': 'no'},
        ]),
        _ev('f1', cliffhanger: 'F', trigger: {
          'day': [0, 0],
        }),
        _ev('h1', cliffhanger: 'H', trigger: {
          'day': [0, 0],
        }),
      ]);
      expect(await run(b, (c) => 0), 'F');
    });

    test('undo after a chained pick: planned event x (also today) keeps inherited rank / is dropped', () async {
      final b = _bundle([
        _ev('m1', layer: 'main', day: 1, cliffhanger: 'M', choices: [
          {'text': 'go', 'next': 'x'},
          {'text': 'no'},
        ]),
        _ev('x', cliffhanger: 'X', trigger: {
          'day': [1, 1],
        }),
      ]);
      // without undo, picking "no": x is a normal planned daily -> main wins.
      expect(await run(b, (c) => 1), 'M');
      // pick "go" then undo and pick "no" -> x should still be planned, and main should win.
      final c = await _ctl(b);
      await c.newGame(seed: 5);
      await _startDay(c);
      final before = c.queuedEventIds.toList();
      final order = <String>[];
      while (c.phase == Phase.event) {
        order.add(c.current!.id);
        if (c.current!.id == 'm1') {
          c.choose(0);
          c.undoChoice();
          c.choose(1);
        } else {
          c.choose(0);
        }
        c.continueAfterChoice();
      }
      print('planned after m1=$before order=$order cliff=${c.cliffhanger}');
      expect(order, contains('x'), reason: 'undo drops a planned event (pre-existing)');
    });

    test('villain bot: always the villain choice when visible; value, endings, combo', () async {
      final b = testBundle();
      bool isVillain(Choice ch) =>
          (ch.effects.stats[Stat.villain] ?? 0) > 0 || (ch.fail.stats[Stat.villain] ?? 0) > 0;
      final endings = <String, int>{};
      final vals = <int>[];
      var villainChoices = 0, villainGood = 0;
      for (final st in b.starts) {
        for (var seed = 0; seed < 8; seed++) {
          SharedPreferences.setMockInitialValues({});
          final c = GameController(bundle: b, save: SaveService());
          await c.init();
          await c.markIntroSeen();
          final rnd = Random(seed);
          await c.newGame(preference: Preference.genders[seed % 2], seed: seed, start: st.id);
          var maxV = 0;
          for (var d = 0; d < 130 && c.ending == null; d++) {
            c.beginMorning();
            c.state!.hearts = b.config.maxHearts;
            // never rest: keeps heat up
            final acts = b.config.actions.where((a) => a.id != 'rest').toList();
            await c.startDay(acts[rnd.nextInt(acts.length)]);
            while (c.phase == Phase.event) {
              final open = [for (final v in c.choices) if (!v.locked) v];
              final v = open.where((v) => isVillain(v.choice)).firstOrNull ??
                  open.where((v) => (v.choice.effects.stats[Stat.heat] ?? 0) > 0).firstOrNull ??
                  open[rnd.nextInt(open.length)];
              final mg = v.choice.minigame != null;
              c.choose(v.index, minigameSuccess: mg ? true : null);
              if (isVillain(v.choice)) {
                villainChoices++;
                if ((c.lastOutcome!.delta.stats[Stat.villain] ?? 0) > 0 && c.lastOutcome!.comboStarted) {
                  villainGood++;
                }
              }
              maxV = max(maxV, c.state!.stat(Stat.villain));
              c.continueAfterChoice();
            }
            if (c.phase == Phase.summary) await c.endDay();
          }
          vals.add(maxV);
          endings.update(c.ending?.id ?? 'none', (x) => x + 1, ifAbsent: () => 1);
        }
      }
      print('villain picks=$villainChoices comboStarted-on-villain=$villainGood');
      print('max villain per run=$vals');
      print('endings=$endings');
    }, timeout: const Timeout(Duration(minutes: 10)));
  });
