// ignore_for_file: avoid_print
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:mossol/ui/settings_screen.dart';
import 'package:mossol/ui/widgets.dart';
import 'helpers.dart';

void main() {
  testWidgets('which settings row overflows at 2.0', (tester) async {
    tester.view.physicalSize = const Size(320, 3000);
    tester.view.devicePixelRatio = 1.0;
    tester.platformDispatcher.textScaleFactorTestValue = 2.0;
    addTearDown(() {
      tester.view.resetPhysicalSize();
      tester.view.resetDevicePixelRatio();
      tester.platformDispatcher.clearTextScaleFactorTestValue();
    });
    final c = await makeController();
    await tester.pumpWidget(wrapApp(SettingsScreen(c: c)));
    await tester.pumpAndSettle();
    tester.takeException();
    for (final e in find.byType(AppListRow).evaluate()) {
      final w = e.widget as AppListRow;
      final rowEl = find.descendant(of: find.byWidget(w), matching: find.byType(Row)).evaluate().first;
      final rf = rowEl.renderObject as RenderFlex;
      var sum = 0.0;
      rf.visitChildren((ch) => sum += (ch as RenderBox).size.width);
      print('${w.title}: row=${rf.size.width} children=$sum ${sum > rf.size.width + 0.5 ? "OVERFLOW" : ""}');
    }
  });
}
