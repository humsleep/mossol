// R1: start pick sheet / heat gauge layout at small screen + large text.
// ignore_for_file: avoid_print
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:mossol/engine/models.dart';
import 'package:mossol/game_controller.dart';
import 'package:mossol/ui/heat_gauge.dart';
import 'package:mossol/ui/start_pick_sheet.dart';
import 'package:mossol/ui/summary_screen.dart';

import 'helpers.dart';

void small(WidgetTester tester, double scale, {Size size = const Size(320, 568)}) {
  tester.view.physicalSize = size;
  tester.view.devicePixelRatio = 1.0;
  tester.platformDispatcher.textScaleFactorTestValue = scale;
  addTearDown(() {
    tester.view.resetPhysicalSize();
    tester.view.resetDevicePixelRatio();
    tester.platformDispatcher.clearTextScaleFactorTestValue();
  });
}

List<String> takeErrors(WidgetTester tester) {
  final out = <String>[];
  Object? e;
  while ((e = tester.takeException()) != null) {
    out.add('$e'.split('\n').first);
  }
  return out;
}

void main() {
  homeMain();
  for (final scale in [1.0, 1.6, 2.0]) {
    testWidgets('start sheet 320x568 scale $scale: no overflow, scroll to every card', (tester) async {
      small(tester, scale);
      final b = testBundle();
      await tester.pumpWidget(wrapApp(Scaffold(
        body: StartPickSheet(
          starts: b.starts,
          endingCount: 0,
          firstRun: true,
          charNames: b.charNames,
          newIds: const {'sc_swap'},
          fateLocked: 'sc_ghost',
        ),
      )));
      await tester.pumpAndSettle();
      final errs = takeErrors(tester);
      for (final s in [...b.starts.map((s) => 'start-card-${s.id}'), 'start-fate']) {
        await tester.scrollUntilVisible(find.byKey(Key(s)), 200,
            scrollable: find.descendant(of: find.byKey(const Key('start-pick-list')), matching: find.byType(Scrollable)));
        await tester.pumpAndSettle();
        errs.addAll(takeErrors(tester));
      }
      print('scale $scale errors: ${errs.toSet()}');
      expect(errs, isEmpty);
    });
  }

  testWidgets('locked card tap does not pop; second tap collapses', (tester) async {
    final b = testBundle();
    StartPick? got;
    var closed = false;
    await tester.pumpWidget(wrapApp(Builder(
      builder: (ctx) => Scaffold(
        body: Center(
          child: TextButton(
            onPressed: () async {
              got = await StartPickSheet.show(ctx, starts: b.starts, endingCount: 0);
              closed = true;
            },
            child: const Text('open'),
          ),
        ),
      ),
    )));
    await tester.tap(find.text('open'));
    await tester.pumpAndSettle();
    final locked = find.byKey(const Key('start-card-sc_swap'));
    await tester.scrollUntilVisible(locked, 200,
        scrollable: find.descendant(of: find.byKey(const Key('start-pick-list')), matching: find.byType(Scrollable)));
    int texts() => find.descendant(of: locked, matching: find.byType(Text)).evaluate().length;
    final before = texts();
    await tester.tap(locked);
    await tester.pumpAndSettle();
    expect(closed, isFalse);
    print('texts before $before after ${texts()}');
    await tester.tap(locked);
    await tester.pumpAndSettle();
    print('texts after 2nd tap ${texts()} closed=$closed got=$got');
    final hook = b.startOf('sc_swap')!.hook;
    print('hook visible after tap: ${find.textContaining(hook.substring(0, 4)).evaluate().length}');
  });

  for (final scale in [1.0, 1.3, 1.6, 2.0]) {
    testWidgets('HeatGauge at summary width, scale $scale, delta+overnight', (tester) async {
      small(tester, scale);
      await tester.pumpWidget(wrapApp(Scaffold(
        body: Padding(
          padding: const EdgeInsets.symmetric(horizontal: 20 + 16),
          child: Column(children: const [
            Align(
              alignment: AlignmentDirectional.centerStart,
              child: HeatGauge(heat: 88, delta: 15, overnight: -1),
            ),
            Align(
              alignment: AlignmentDirectional.centerStart,
              child: HeatGauge(heat: 45),
            ),
          ]),
        ),
      )));
      await tester.pumpAndSettle();
      final errs = takeErrors(tester);
      print('heat scale $scale: $errs');
      expect(errs, isEmpty);
    });
  }

  for (final (w, h, scale) in [(320.0, 568.0, 1.0), (320.0, 568.0, 1.3), (375.0, 667.0, 1.0), (375.0, 667.0, 1.3), (390.0, 844.0, 1.3), (390.0, 844.0, 1.6)]) {
    testWidgets('SummaryScreen real $w x $h with high heat scale $scale', (tester) async {
      small(tester, scale, size: Size(w, h));
      final c = await makeController();
      await c.newGame(preference: Preference.female, seed: 1, start: 'sc_clip');
      c.beginMorning();
      await c.startDay(c.config.actions.first);
      while (c.phase == Phase.event) {
        c.choose(plainChoiceIndex(c));
        c.continueAfterChoice();
      }
      c.state!.stats[Stat.heat] = 90;
      c.dayDelta.stats[Stat.heat] = 25;
      await tester.pumpWidget(wrapApp(SummaryScreen(c: c)));
      await tester.pump(const Duration(seconds: 3));
      final g = find.byType(HeatGauge);
      if (g.evaluate().isNotEmpty) { await tester.ensureVisible(g); await tester.pump(const Duration(seconds: 2)); }
      print('gauge found ${g.evaluate().length}');
      final errs = takeErrors(tester);
      print('summary $w scale $scale: $errs');
      expect(errs, isEmpty);
    });
  }
}

void homeMain() {
  for (final (scale, heat) in [(1.0, 90), (1.6, 90), (2.0, 90), (2.0, 0)]) {
    testWidgets('Home with heat $heat at 320 scale $scale', (tester) async {
      small(tester, scale);
      final c = await makeController();
      await c.newGame(preference: Preference.female, seed: 1, start: heat == 0 ? 'classic' : 'sc_clip');
      c.state!.stats[Stat.heat] = heat;
      c.goHome();
      await tester.pumpWidget(fullApp(c));
      await tester.pump(const Duration(seconds: 2));
      final g = find.byType(HeatGauge);
      print('home gauge ${g.evaluate().length}');
      final errs = takeErrors(tester);
      if (errs.isNotEmpty) debugDumpApp();
      print('home scale $scale heat $heat: $errs');
      expect(errs, isEmpty);
    });
  }
}
