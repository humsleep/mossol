Map<String, Object> _readDump(String name) {
  final j = jsonDecode(File('$_dumps/$name.json').readAsStringSync()) as Map;
  return {
    for (final e in j.entries)
      e.key as String: e.value is List
          ? (e.value as List).cast<String>().toList()
          : e.value as Object,
  };
}

Future<Map<String, Object?>> _prefs() async {
  final p = await SharedPreferences.getInstance();
  return {for (final k in p.getKeys()) k: p.get(k)};
}

/// Plays the current event like the screen does (asks the clock first), returns the event id.
String _playOne(GameController c, Random rnd, {List<String>? problems}) {
  final ev = c.current!;
  final s = c.state!;
  if (ev.lines.isEmpty && ev.choices.isEmpty) problems?.add('empty event ${ev.id} D${s.day}');
  if (!ev.isCall) {
    c.clockStartFor(
      ev,
      ChatClock.startSeconds(
        seed: s.seed,
        day: s.day,
        index: c.todayEventIndex,
        total: c.todayEventTotal,
      ),
    );
  }
  final open = [for (final v in c.choices) if (!v.locked) v.index];
  if (open.isEmpty) {
    problems?.add('no open choice ${ev.id} D${s.day}');
    return ev.id;
  }
  final i = open[rnd.nextInt(open.length)];
  final mg = ev.choices[i].minigame != null;
  c.choose(i, minigameSuccess: mg ? rnd.nextBool() : null);
  c.continueAfterChoice();
  return ev.id;
}

Future<void> _playToEnd(
  GameController c,
  Random rnd, {
  List<String>? problems,
  int maxDays = 130,
}) async {
  final b = c.bundle;
  for (var d = 0; d < maxDays && c.ending == null; d++) {
    if (c.phase == Phase.dayStart) c.beginMorning();
    if (c.phase == Phase.action) {
      c.state!.hearts = b.config.maxHearts;
      await c.startDay(b.config.actions[rnd.nextInt(b.config.actions.length)]);
    }
    var guard = 0;
    while (c.phase == Phase.event && guard++ < 40) {
      _playOne(c, rnd, problems: problems);
    }
    if (guard >= 40) problems?.add('event loop D${c.state!.day}');
    if (c.phase == Phase.summary) await c.endDay();
  }
}

const _config = '{"initialStats": {"charm": 10}, "actions": []}';
const _chars = '''[
  {"id": "a", "name": "가람", "gender": "f", "role": "senior"},
  {"id": "b", "name": "나래", "gender": "m", "role": "senior"}
]''';
const _endings =
    '[{"id": "solo", "name": "솔로", "tier": "solo", "default": true}]';

Map<String, dynamic> _ev(
  String id, {
  String layer = 'daily',
  int? day,
  Map<String, dynamic>? trigger,
  List<Map<String, dynamic>>? lines,
  List<Map<String, dynamic>>? choices,
  String? cliffhanger,
  bool? once,
}) => {
  'id': id,
  'layer': layer,
  'day': ?day,
  'trigger': ?trigger,
  'cliffhanger': ?cliffhanger,
  'once': ?once,
  'lines':
      lines ??
      [
        {'who': 'them', 'text': '안녕'},
        {'who': 'them', 'text': '둘째 줄'},
      ],
  'choices':
      choices ??
      [
        {'text': '하나'},
        {'text': '둘'},
      ],
};

StoryBundle _bundle(List<Map<String, dynamic>> events) =>
    StoryBundle.fromJsonStrings(
      config: _config,
      characters: _chars,
      events: [jsonEncode(events)],
      endings: _endings,
    );

  group('live build (HEAD) saves', () {

      test('$name: init → migration → continue → play to ending', () async {
        final dump = _readDump(name);
        SharedPreferences.setMockInitialValues(dump);
        final save0 = jsonDecode(dump['mossol_save_v1'] as String) as Map<String, dynamic>;
        final meta0 = jsonDecode(dump['mossol_meta_v1'] as String) as Map<String, dynamic>;
        final album = (dump['mossol_endings_v1'] as List<String>?) ?? const <String>[];

        final c = GameController(bundle: testBundle(), save: SaveService());

        final m = c.meta!;
        if (album.isNotEmpty) {
          expect(m.completedStarts, [StartScenario.classic]);
          expect(m.startEndings[StartScenario.classic]!.toSet(), album.toSet());
        } else {
          expect(m.completedStarts, isEmpty);
        }
        if ((meta0['totalRuns'] as int) > 1) {
          expect(m.seenEvents.toSet(), (save0['seen'] as List).cast<String>().toSet());
        } else {
          expect(m.seenEvents, isEmpty);
        }
        // idempotent: a second launch does not change meta
        final metaAfter1 = (await _prefs())['mossol_meta_v1'];
        final c2 = GameController(bundle: testBundle(), save: SaveService());
        await c2.init();
        expect((await _prefs())['mossol_meta_v1'], metaAfter1);

        expect(await c2.continueGame(), isTrue);
        print('$name: phase=${c2.phase} day=${c2.state!.day} cur=${c2.current?.id} '
            'cliff=${c2.cliffhanger} clockEnd=${c2.dayClockEnd} ff=${c2.canFastForward} '
            'suggested=${c2.suggestedStart}');
        final problems = <String>[];
        await _playToEnd(c2, Random(name.hashCode), problems: problems);
        expect(problems, isEmpty);
        expect(c2.ending, isNotNull);
        print('$name: ending=${c2.ending!.id} completed=${c2.completedStarts} '
            'seenEvents=${c2.meta!.seenEvents.length}');
        // next run: veteran/done flags

      }, timeout: const Timeout(Duration(minutes: 3)));

  // ---------------------------------------------------------------------------
  group('variant-seen tracking', () {
    test('absorbed id#index == variants actually shown (with random mid-day restarts)', () async {
      final b = testBundle();
      var mismatches = 0;
      var checked = 0;
      var ffWrong = 0;
      var ffChecks = 0;
      for (var seed = 0; seed < 6; seed++) {
        SharedPreferences.setMockInitialValues({});
        var c = GameController(bundle: b, save: SaveService());
        await c.init();
        await c.markIntroSeen();
        final rnd = Random(seed);
        final st = b.starts[seed % b.starts.length].id;
        await c.newGame(start: st, seed: seed + 100, preference: Preference.genders[seed % 2]);
        final shown = <String>{};
        for (var d = 0; d < 130 && c.ending == null; d++) {
          if (c.phase == Phase.dayStart) c.beginMorning();
          c.state!.hearts = 5;
          await c.startDay(b.config.actions[rnd.nextInt(b.config.actions.length)]);
          while (c.phase == Phase.event) {
            final ev = c.current!;
            final orig = b.eventById[ev.id]!;
            if (orig.variants.isNotEmpty) {
              // which bundle is on screen? compare the lines
              final s = c.state!;
              final idx = c.engine.variantIndexOf(s, orig);
              shown.add('${ev.id}#$idx');
            }
            _playOne(c, rnd);
            if (rnd.nextInt(10) == 0 && c.phase == Phase.event) {
              // relaunch mid-day
              c = GameController(bundle: b, save: SaveService());
              await c.init();
              await c.continueGame();
            }
          }
          if (c.phase == Phase.summary) await c.endDay();
        }
        final keys = c.meta!.seenEvents.where((k) => k.contains('#')).toSet();
        checked += shown.length;
        final missing = shown.difference(keys);
        final extra = keys.difference(shown);
        mismatches += missing.length + extra.length;
        if (missing.isNotEmpty || extra.isNotEmpty) {
          print('seed $seed: missing=$missing extra=$extra');
        }
        // run 2: fast-forward must only claim variants that were shown
        await c.nextRun(start: st);
        final shown2 = <String>{};
        for (var d = 0; d < 40 && c.ending == null; d++) {
          if (c.phase == Phase.dayStart) c.beginMorning();
          c.state!.hearts = 5;
          await c.startDay(b.config.actions[rnd.nextInt(b.config.actions.length)]);
          while (c.phase == Phase.event) {
            final ev = c.current!;
            final orig = b.eventById[ev.id]!;
            if (orig.variants.isNotEmpty) {
              final idx = c.engine.variantIndexOf(c.state!, orig);
              final key = '${ev.id}#$idx';
              ffChecks++;
              if (c.canFastForward && !shown.contains(key) && !shown2.contains(key)) {
                ffWrong++;
                print('ff on unseen variant $key');
              }
              shown2.add(key);
            }
            _playOne(c, rnd);
          }
          if (c.phase == Phase.summary) await c.endDay();
        }
      }
      print('variant keys checked=$checked mismatches=$mismatches ffChecks=$ffChecks ffWrong=$ffWrong');
      expect(mismatches, 0);
      expect(ffWrong, 0);
    }, timeout: const Timeout(Duration(minutes: 8)));
  });

              'ripples=${c.meta!.seenRipples.length} save=${p.getString('mossol_save_v1')?.length}B');
        }

      final evs = b.events.length;
      final varKeys = b.events.fold<int>(0, (a, e) => a + (e.variants.isEmpty ? 0 : e.variants.length + 1));
